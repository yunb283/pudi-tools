SQLite 表结构（v1）WAL + 外键级联
PRAGMA journal_mode = WAL;
PRAGMA synchronous = NORMAL;
PRAGMA foreign_keys = ON;

-- 一次扫描 = 一份快照
CREATE TABLE snapshots (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  root_path    TEXT    NOT NULL DEFAULT 'C:\', -- v1 固定，不开放选择
  started_at   INTEGER NOT NULL,        -- Unix ms
  finished_at  INTEGER,
  status       TEXT    NOT NULL,        -- running/complete/partial/cancelled/failed
  file_count   INTEGER NOT NULL DEFAULT 0,
  dir_count    INTEGER NOT NULL DEFAULT 0,
  total_bytes  INTEGER NOT NULL DEFAULT 0, -- 递归汇总，非条目累加
  error_count  INTEGER NOT NULL DEFAULT 0,
  is_auto      INTEGER NOT NULL DEFAULT 0,
  note         TEXT
);

-- 条目：文件原始大小 + 目录汇总大小；path 为规范化绝对/相对键
CREATE TABLE entries (
  snapshot_id  INTEGER NOT NULL REFERENCES snapshots(id) ON DELETE CASCADE,
  path         TEXT    NOT NULL,
  parent_path  TEXT,                   -- 建树用
  name         TEXT    NOT NULL,
  kind         TEXT    NOT NULL,        -- dir / file
  size_bytes   INTEGER NOT NULL,        -- 目录=子树汇总
  own_bytes    INTEGER NOT NULL DEFAULT 0, -- 目录直属文件大小
  file_count   INTEGER NOT NULL DEFAULT 0,
  ext          TEXT,                    -- 文件类型
  mtime        INTEGER,
  is_reparse   INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (snapshot_id, path)
);
CREATE INDEX idx_entries_parent ON entries(snapshot_id, parent_path);
CREATE INDEX idx_entries_size   ON entries(snapshot_id, size_bytes DESC);

-- 扫描异常：权限不足 / 文件消失 / IO / 重解析点跳过
CREATE TABLE scan_errors (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  snapshot_id  INTEGER NOT NULL REFERENCES snapshots(id) ON DELETE CASCADE,
  path         TEXT NOT NULL,
  kind         TEXT NOT NULL,
  message      TEXT
);

-- 键值配置：最大快照数 / 保留天数 / 线程数 / 批量条数
CREATE TABLE settings ( key TEXT PRIMARY KEY, value TEXT NOT NULL );

-- 计划任务登记（实际任务在 Windows 任务计划程序里）
CREATE TABLE schedules (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  name       TEXT NOT NULL UNIQUE,
  kind       TEXT NOT NULL,     -- daily / weekly
  time_of_day TEXT NOT NULL,    -- HH:MM
  weekday    INTEGER,
  enabled    INTEGER NOT NULL DEFAULT 1,
  task_name  TEXT
);
关键算法与正确性约束
01 扫描根目录恒为 C:\；jwalk 并行遍历，follow_links(false)；只取 DirEntry 元数据，绝不开文件；重解析点直接 prune，防止循环与重复计数。AtomicBool 取消标志每个条目检查一次。
02 汇总按路径深度倒序做后序聚合：dir.size = 直属文件大小 + Σ 子目录已聚合 size。每个文件只归属其直接父目录一次；根的 total_bytes 取汇总结果，而非把 entries 全部相加。
03 写入后台线程消费通道，每 2,000 条一个事务；进度事件 100ms 节流推送；取消时回滚当前事务并把快照标记 cancelled，不落半份数据。
04 对比以规范化 path 做全外连接，得到 added / removed / grown / shrunk / unchanged；pct = (B-A)/A，A=0 时显示「新增」而非无穷大；任一快照非 complete 时强制显示警示、隐藏精确百分比、禁用精确结论导出。
05 安全不读内容、不上传、不自动删除系统文件；「删除快照」只删 DB 记录并需二次确认；资源管理器打开只读调用 explorer /select。
测试计划验证目录大小统计与快照对比正确性
T1aggregate.rs：3 层嵌套 + 空目录 + 同名目录 + 直属文件，断言每个目录汇总值等于手工计算；根汇总 = 所有直属项之和；硬链接 / 重解析点不被重复计入。
T2diff.rs：构造覆盖五态的路径集合，断言 delta / pct / status；A=0 时百分比语义；路径大小写与分隔符规范化后能正确配对。
T3集成：测试夹具用临时目录（仅测试代码，产品路径固定 C:\）建树 → 扫描 A → 增/删/改文件 → 扫描 B → 断言 Top-N 增长目录及字节数与预期完全一致，且父目录变化不被子目录重复放大。
T4异常：注入权限不足与文件消失，断言错误落库、快照状态为 partial、对比接口拒绝给出精确结论。
T5前端 Vitest：字节/日期格式化、树展开折叠、三种排序与筛选逻辑；取消扫描时进度事件停止。
T6大规模夹具（10 万小文件）验证批量写入耗时与内存上限，避免超大目录 OOM。
